CREATE TABLE `user6_form` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `name` VARCHAR(100) NOT NULL,
    `email` VARCHAR(255) NOT NULL UNIQUE,
    `password` VARCHAR(255) NOT NULL,
    `contact_number` VARCHAR(15) NULL,
    `linkedin_id` VARCHAR(255) NULL,
    `portfolio` VARCHAR(255) NULL,
    `job_title` VARCHAR(100) NULL,
    `professional_summary` TEXT NULL,
    `work_experience` TEXT NULL,
    `hard_skills` TEXT NULL,
    `soft_skills` TEXT NULL,
    `certifications` TEXT NULL,
    `education` TEXT NULL,
    `projects` TEXT NULL,
    `languages` TEXT NULL,
    `professional_affiliations` TEXT NULL,
    `volunteer_experience` TEXT NULL,
    `publications` TEXT NULL,
    `awards` TEXT NULL,
    `image` VARCHAR(255) NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
